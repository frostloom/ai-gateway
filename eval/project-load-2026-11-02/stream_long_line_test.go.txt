 const chunks=2048
 chunk:=strings.Repeat("x",8192);release:=make(chan struct{});sent:=make(chan struct{});done:=make(chan struct{})
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){_,_=io.Copy(io.Discard,r.Body);w.Header().Set("Content-Type","text/event-stream");_,_=io.WriteString(w,"data: ");for i:=0;i<chunks;i++{if _,err:=io.WriteString(w,chunk);err!=nil{return}};w.(http.Flusher).Flush();close(sent);<-release;_,_=io.WriteString(w,"\n\ndata: [DONE]\n\n")}));defer server.Close()
 writer:=&streamDiscardWriter{header:make(http.Header),target:1<<60,reached:make(chan struct{})};c,_:=gin.CreateTestContext(writer);c.Request=httptest.NewRequest(http.MethodPost,"/v1/chat/completions",nil);bill:=&fakeBilling{}

 go func(){relayStream(c,bill,&fakeRouter{},slog.New(slog.DiscardHandler),&client.RouteResult{ProviderID:1,BaseURL:server.URL},&chatRequest{Model:"m",Messages:[]chatMessage{{Role:"user",Content:"hi"}},Stream:true},"line-memory",1,1,true);close(done)}()

}
