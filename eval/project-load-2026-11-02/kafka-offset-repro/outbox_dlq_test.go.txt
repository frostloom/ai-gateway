// Real Store and MySQL; producers model a failed main publish and accepted DLQ.
func TestIsolatedDLQMarkFailureRemainsRetryable(t *testing.T) {
 dsn:=os.Getenv("ISOLATED_MYSQL_DSN"); if dsn=="" {t.Fatal("isolated MySQL required")}
 db,err:=gorm.Open(mysql.Open(dsn),&gorm.Config{}); if err!=nil {t.Fatal(err)}
 if err=db.AutoMigrate(&store.OutboxEvent{}); err!=nil {t.Fatal(err)}
 row:=evRow(1,"isolated:settle",4); if err=db.Create(&row).Error; err!=nil {t.Fatal(err)}
 var injected atomic.Bool
 if err=db.Callback().Update().Before("gorm:update").Register("test:fail-dlq-mark",func(tx *gorm.DB) {
  if _,ok:=tx.Statement.Dest.(map[string]any)["topic"]; ok && injected.CompareAndSwap(false,true) {tx.AddError(errors.New("injected dlq mark failure"))}
 });err!=nil {t.Fatal(err)}
 st:=store.NewStore(db,nil); main:=&stubProducer{fail:map[string]error{row.EventKey:errBoom}}; dlq:=&stubProducer{}
 relay:=newRelay(st,main,dlq,5); ctx:=context.Background()
 if _,err=relay.FlushOnce(ctx);err!=nil {t.Fatal(err)}
 if !injected.Load() {t.Fatal("mark failure not injected")}
 pending,err:=st.PollOutbox(ctx,10,5);if err!=nil {t.Fatal(err)}
 if len(pending)!=1 {var actual store.OutboxEvent; db.First(&actual,row.ID);t.Fatalf("failed DLQ mark must remain pollable, got %d; sent=%v attempts=%d",len(pending),actual.SentAt,actual.Attempts)}
 if _,err=relay.FlushOnce(ctx);err!=nil {t.Fatal(err)}
 var result store.OutboxEvent; if err=db.First(&result,row.ID).Error;err!=nil {t.Fatal(err)}
 if result.SentAt==nil || result.Topic!=store.TopicBillingDLQ || len(dlq.ok)!=2 {t.Fatalf("recovery failed: row=%+v dlq publishes=%d",result,len(dlq.ok))}
}

type simultaneousPoll struct { Store; seen chan struct{}; release chan struct{} }
func (s *simultaneousPoll) PollOutbox(ctx context.Context, limit,max int)([]store.OutboxEvent,error) {
 rows,err:=s.Store.PollOutbox(ctx,limit,max); s.seen<-struct{}{}; <-s.release; return rows,err
}

func TestIsolatedTwoRelaysDoNotStrandUnsent(t *testing.T) {
 dsn:=os.Getenv("ISOLATED_MYSQL_DSN"); if dsn=="" {t.Fatal("isolated MySQL required")}
 db,err:=gorm.Open(mysql.Open(dsn),&gorm.Config{});if err!=nil {t.Fatal(err)}
 if err=db.AutoMigrate(&store.OutboxEvent{});err!=nil {t.Fatal(err)}
 row:=evRow(2,"concurrent:settle",3);if err=db.Create(&row).Error;err!=nil {t.Fatal(err)}
 st:=store.NewStore(db,nil); barrier:=&simultaneousPoll{Store:st,seen:make(chan struct{},2),release:make(chan struct{})}
 main:=&stubProducer{fail:map[string]error{row.EventKey:errBoom}}; dlq:=&stubProducer{}
 var wg sync.WaitGroup; results:=make(chan error,2)
 for i:=0;i<2;i++ {wg.Add(1);go func(){defer wg.Done(); _,err:=newRelay(barrier,main,dlq,5).FlushOnce(context.Background());results<-err}()}
 <-barrier.seen; <-barrier.seen; close(barrier.release);wg.Wait();close(results)
 for err:=range results {if err!=nil {t.Fatal(err)}}
 var actual store.OutboxEvent;if err=db.First(&actual,row.ID).Error;err!=nil {t.Fatal(err)}
 pending,err:=st.PollOutbox(context.Background(),10,5);if err!=nil {t.Fatal(err)}
 if actual.SentAt==nil && len(pending)==0 {t.Fatalf("concurrent relays stranded unsent row: attempts=%d, dlq publishes=%d",actual.Attempts,len(dlq.ok))}
 if _,err=newRelay(st,main,dlq,5).FlushOnce(context.Background());err!=nil {t.Fatal(err)}
 if err=db.First(&actual,row.ID).Error;err!=nil {t.Fatal(err)}
 if actual.SentAt==nil || actual.Topic!=store.TopicBillingDLQ || len(dlq.ok)!=1 {t.Fatalf("concurrent row did not recover to DLQ: row=%+v publishes=%d",actual,len(dlq.ok))}
}

 first:=evRow(3,"blocked",4); second:=evRow(4,"later",0)

 main:=&stubProducer{fail:map[string]error{"blocked":errBoom}};dlq:=&stubProducer{fail:map[string]error{"blocked":errBoom}}
 relay:=NewRelay(store.NewStore(db,nil),main,dlq,Config{BatchSize:1,MaxAttempts:5},logger.New("error"))
