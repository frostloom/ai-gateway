type mysqlOutageGate struct { Producer; first atomic.Bool; confirmed chan struct{}; release chan struct{} }
func (g *mysqlOutageGate) Publish(ctx context.Context,key string,value []byte) error {
	if g.first.CompareAndSwap(false,true) {
		if err:=g.Producer.Publish(ctx,key,value); err!=nil { return err }
		close(g.confirmed)
		select { case <-g.release: return nil; case <-ctx.Done(): return ctx.Err() }
	}
	return g.Producer.Publish(ctx,key,value)
}

func TestIsolatedRelayMySQLMarkOutage(t *testing.T) {
	dsn,addr,name:=os.Getenv("ISOLATED_MYSQL_DSN"),os.Getenv("ISOLATED_KAFKA_ADDR"),os.Getenv("ISOLATED_MYSQL_CONTAINER")
	if !regexp.MustCompile(`^relay-experiment-[0-9a-f]{12}-mysql$`).MatchString(name) { t.Fatal("refusing non-test mysql container") }
	cfg,err:=mysqldriver.ParseDSN(dsn); if err!=nil || !strings.HasPrefix(cfg.Addr,"127.0.0.1:") { t.Fatalf("invalid test DSN: %v",err) }
	port,err:=exec.Command("docker","port",name,"3306/tcp").CombinedOutput(); if err!=nil || strings.TrimSpace(string(port))!=cfg.Addr { t.Fatalf("DSN/port mismatch=%s err=%v",port,err) }
	db,err:=gorm.Open(mysql.Open(dsn),&gorm.Config{}); if err!=nil { t.Fatal(err) }
	if err:=db.AutoMigrate(&store.OutboxEvent{},&store.AuditEvent{}); err!=nil { t.Fatal(err) }
	const count=16
	prefix:=fmt.Sprintf("mysql-outage-%d-",time.Now().UnixNano())
	for i:=0;i<count;i++ { id:=fmt.Sprintf("%s%d",prefix,i); payload,_:=json.Marshal(store.BillingEvent{EventID:id,RequestID:id,Phase:"reserve",TenantID:1,Funding:"balance"}); if err:=db.Create(&store.OutboxEvent{Topic:store.TopicBillingEvents,EventKey:id,Payload:string(payload),CreatedAt:time.Now()}).Error; err!=nil { t.Fatal(err) } }
	log:=logger.New("error"); base:=store.NewStore(db,nil); prod:=appkafka.NewProducer([]string{addr},"offset-repro",log); defer prod.Close()
	gate:=&mysqlOutageGate{Producer:prod,confirmed:make(chan struct{}),release:make(chan struct{})}
	relay:=NewRelay(base,gate,prod,Config{BatchSize:count,MaxAttempts:5},log)
	ctx,cancel:=context.WithTimeout(context.Background(),15*time.Second)
	type outcome struct{ stats Stats; err error }; result:=make(chan outcome,1)
	go func(){ stats,e:=relay.FlushOnce(ctx); result<-outcome{stats,e} }()
	select { case <-gate.confirmed: case <-ctx.Done(): cancel(); t.Fatal("first publish not confirmed") }
	callCtx,done:=context.WithTimeout(context.Background(),30*time.Second); defer done()
	if out,e:=exec.CommandContext(callCtx,"docker","kill",name).CombinedOutput(); e!=nil { cancel(); t.Fatalf("kill: %v %s",e,out) }
	restarted:=false
	t.Cleanup(func(){ if !restarted { _=exec.Command("docker","start",name).Run() } })
	close(gate.release)
	first:=<-result; cancel()
	if first.err!=nil || first.stats.Sent<1 || first.stats.MarkSentErrors<1 { t.Fatalf("outage stats=%+v err=%v",first.stats,first.err) }
	w:=httptest.NewRecorder(); metrics.Handler().ServeHTTP(w,httptest.NewRequest("GET","/metrics",nil))
	if w.Code!=200 || !strings.Contains(w.Body.String(),fmt.Sprintf("billing_outbox_mark_errors_total{operation=\"sent\"} %d\n",first.stats.MarkSentErrors)) { t.Fatalf("counter mismatch status=%d errors=%d body=%s",w.Code,first.stats.MarkSentErrors,w.Body.String()) }
	if out,e:=exec.CommandContext(callCtx,"docker","start",name).CombinedOutput(); e!=nil { t.Fatalf("start: %v %s",e,out) }; restarted=true
	sqlDB,err:=db.DB(); if err!=nil { t.Fatal(err) }
	deadline:=time.Now().Add(90*time.Second)
	for { pingCtx,stop:=context.WithTimeout(context.Background(),time.Second); err=sqlDB.PingContext(pingCtx); stop(); if err==nil {break}; if time.Now().After(deadline) { t.Fatalf("mysql restart: %v",err) }; time.Sleep(200*time.Millisecond) }
	var unsent int64
	if err:=db.Model(&store.OutboxEvent{}).Where("event_key LIKE ? AND sent_at IS NULL",prefix+"%").Count(&unsent).Error; err!=nil || unsent!=count { t.Fatalf("restart unsent=%d err=%v",unsent,err) }
	consumerCtx,stop:=context.WithCancel(context.Background()); kc:=appkafka.NewConsumer([]string{addr},"mysql-outage-group",[]string{"offset-repro"}); c:=consumer.New(base,kc,consumer.Config{},log); finished:=make(chan struct{})
	go func(){ defer close(finished); _=c.Run(consumerCtx) }(); defer func(){ stop(); <-finished; _=kc.Close() }()
	deadline=time.Now().Add(60*time.Second); var audit int64
	for time.Now().Before(deadline) { flushCtx,end:=context.WithTimeout(context.Background(),5*time.Second); _,err:=relay.FlushOnce(flushCtx); end(); if err!=nil {t.Fatal(err)}; if err:=db.Model(&store.OutboxEvent{}).Where("event_key LIKE ? AND sent_at IS NULL",prefix+"%").Count(&unsent).Error; err!=nil {t.Fatal(err)}; if err:=db.Model(&store.AuditEvent{}).Where("event_id LIKE ?",prefix+"%").Count(&audit).Error; err!=nil {t.Fatal(err)}; if unsent==0 && audit==count && c.Stats().Consumed>=count+1 { break }; time.Sleep(300*time.Millisecond) }
	if unsent!=0 || audit!=count || c.Stats().Consumed<count+1 { t.Fatalf("recovery unsent=%d audit=%d consumed=%d",unsent,audit,c.Stats().Consumed) }
	var mismatch int64
	if err:=db.Raw("SELECT COUNT(*) FROM outbox_events o LEFT JOIN audit_events a ON a.event_id=o.event_key WHERE o.event_key LIKE ? AND (a.id IS NULL OR NOT (o.payload <=> a.payload))",prefix+"%").Scan(&mismatch).Error; err!=nil || mismatch!=0 {t.Fatalf("mismatch=%d err=%v",mismatch,err)}
	t.Logf("first published before MySQL kill; mark_errors=%d, restart unsent=16, final unsent=%d audit=%d consumed=%d json_mismatch=0",first.stats.MarkSentErrors,unsent,audit,c.Stats().Consumed)
}
