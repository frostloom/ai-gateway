 var writers [streams]*streamDiscardWriter;var bills [streams]*fakeBilling;done:=make(chan struct{},streams)

 for i:=0;i<streams;i++ {writers[i]=&streamDiscardWriter{header:make(http.Header),target:int64(len(frame)*chunks),reached:make(chan struct{})};bills[i]=&fakeBilling{};c,_:=gin.CreateTestContext(writers[i]);c.Request=httptest.NewRequest(http.MethodPost,"/v1/chat/completions",nil);bill:=bills[i]

 }
 for _,w:=range writers {select{case <-w.reached:case <-time.After(45*time.Second):close(release);for i:=0;i<streams;i++{<-done};t.Fatal("streams did not reach target")}}

 retained:=int64(during.HeapAlloc)-int64(before.HeapAlloc);t.Logf("streams=%d bytes_each=%d retained_heap_delta=%d baseline=%t",streams,len(frame)*chunks,retained,os.Getenv("TEST_STREAM_BASELINE")=="1")
 if os.Getenv("TEST_STREAM_BASELINE")=="1" {if retained<24<<20{t.Fatalf("baseline did not retain expected data: %d",retained)}}else if retained>16<<20{t.Fatalf("candidate retained excessive data: %d",retained)}
}
