
func TestRealRouterFullQueueRejectedExplicitly(t *testing.T) {
 dsn:=os.Getenv("COMPARISON_MYSQL_DSN");if dsn==""{t.Fatal("isolated MySQL required")};db,e:=gorm.Open(mysql.Open(dsn),&gorm.Config{Logger:logger.Default.LogMode(logger.Silent)});if e!=nil{t.Fatal(e)}
 if e=db.Exec("CREATE TABLE IF NOT EXISTS providers (id BIGINT PRIMARY KEY,status TINYINT NOT NULL,consecutive_fail INT NOT NULL)").Error;e!=nil{t.Fatal(e)};if e=db.Exec("DELETE FROM providers").Error;e!=nil{t.Fatal(e)};if e=db.Exec("INSERT INTO providers(id,status,consecutive_fail) VALUES(1,0,0)").Error;e!=nil{t.Fatal(e)}
 r:=New(db,slog.New(slog.DiscardHandler));api,ok:=any(r).(interface{testTryReport(uint64,bool)(<-chan struct{},error);CloseReportWorker(context.Context)error});if !ok{t.Fatal("bounded lifecycle unavailable")}
 entered,resume:=make(chan struct{}),make(chan struct{});var once atomic.Bool
 if e=db.Callback().Update().Before("gorm:update").Register("test:full-http",func(tx *gorm.DB){if tx.Statement.Table=="providers"&&once.CompareAndSwap(false,true){close(entered);<-resume}});e!=nil{t.Fatal(e)};defer db.Callback().Update().Remove("test:full-http")
 first,e:=api.testTryReport(1,true);if e!=nil{t.Fatal(e)};select{case <-entered:case <-time.After(5*time.Second):close(resume);t.Fatal("DB callback not reached")}
 accepted:=make([]<-chan struct{},0,1024);for i:=0;i<1024;i++{ch,e:=api.testTryReport(1,true);if e!=nil{close(resume);t.Fatalf("fill %d: %v",i,e)};accepted=append(accepted,ch)}
 req:=httptest.NewRequest(http.MethodPost,"/report",strings.NewReader(`{"provider_id":1,"ok":false}`));req.Header.Set("X-Control-Token","isolated-token");w:=httptest.NewRecorder();NewHTTPHandler(r,"isolated-token",slog.New(slog.DiscardHandler)).ServeHTTP(w,req)
 if w.Code!=http.StatusServiceUnavailable || w.Header().Get("X-Test-Report-Outcome")!="not-accepted"{close(resume);t.Fatalf("want explicit rejection status=503 got=%d marker=%q",w.Code,w.Header().Get("X-Test-Report-Outcome"))}
 close(resume);for _,ch:=range append(accepted,first){select{case <-ch:case <-time.After(5*time.Second):t.Fatal("queue did not drain")}}
 ctx,cancel:=context.WithTimeout(context.Background(),5*time.Second);defer cancel();if e:=api.CloseReportWorker(ctx);e!=nil{t.Fatal(e)}
 states:=r.States();if len(states)!=1||states[0].ConsecutiveFail!=0{t.Fatalf("expected dropped failure, states=%+v",states)}
}

func TestRealRouterAcceptedTimeoutIsUnknown(t *testing.T) {
 dsn:=os.Getenv("COMPARISON_MYSQL_DSN");if dsn==""{t.Fatal("isolated MySQL required")};db,e:=gorm.Open(mysql.Open(dsn),&gorm.Config{Logger:logger.Default.LogMode(logger.Silent)});if e!=nil{t.Fatal(e)}
 if e=db.Exec("CREATE TABLE IF NOT EXISTS providers (id BIGINT PRIMARY KEY,status TINYINT NOT NULL,consecutive_fail INT NOT NULL)").Error;e!=nil{t.Fatal(e)};if e=db.Exec("DELETE FROM providers").Error;e!=nil{t.Fatal(e)};if e=db.Exec("INSERT INTO providers(id,status,consecutive_fail) VALUES(1,0,0)").Error;e!=nil{t.Fatal(e)}
 r:=New(db,slog.New(slog.DiscardHandler));api,ok:=any(r).(interface{testTryReport(uint64,bool)(<-chan struct{},error);CloseReportWorker(context.Context)error});if !ok{t.Fatal("bounded lifecycle unavailable")}
 entered,resume:=make(chan struct{}),make(chan struct{});var once atomic.Bool
 if e=db.Callback().Update().Before("gorm:update").Register("test:unknown-http",func(tx *gorm.DB){if tx.Statement.Table=="providers"&&once.CompareAndSwap(false,true){close(entered);<-resume}});e!=nil{t.Fatal(e)};defer db.Callback().Update().Remove("test:unknown-http")
 first,e:=api.testTryReport(1,true);if e!=nil{t.Fatal(e)};select{case <-entered:case <-time.After(5*time.Second):close(resume);t.Fatal("DB callback not reached")}
 ctx,cancel:=context.WithTimeout(context.Background(),30*time.Millisecond);defer cancel();req:=httptest.NewRequest(http.MethodPost,"/report",strings.NewReader(`{"provider_id":1,"ok":false}`)).WithContext(ctx);req.Header.Set("X-Control-Token","isolated-token");w:=httptest.NewRecorder();NewHTTPHandler(r,"isolated-token",slog.New(slog.DiscardHandler)).ServeHTTP(w,req)
 if w.Code!=http.StatusServiceUnavailable||w.Header().Get("X-Test-Report-Outcome")!="unknown"{close(resume);t.Fatalf("expected unknown response, status=%d marker=%q",w.Code,w.Header().Get("X-Test-Report-Outcome"))}
 close(resume);select{case <-first:case <-time.After(5*time.Second):t.Fatal("first stuck")};done,cancelDone:=context.WithTimeout(context.Background(),5*time.Second);defer cancelDone();if e:=api.CloseReportWorker(done);e!=nil{t.Fatal(e)}
 states:=r.States();if len(states)!=1||states[0].ConsecutiveFail!=1{t.Fatalf("accepted failure was not later applied: %+v",states)}
}

func TestRealRouterCandidateAcknowledgesDatabaseWriteFailure(t *testing.T) {
 dsn:=os.Getenv("COMPARISON_MYSQL_DSN");if dsn==""{t.Fatal("isolated MySQL required")};db,e:=gorm.Open(mysql.Open(dsn),&gorm.Config{Logger:logger.Default.LogMode(logger.Silent)});if e!=nil{t.Fatal(e)}
 if e:=db.Exec("DROP TABLE IF EXISTS providers").Error;e!=nil{t.Fatal(e)}
 r:=New(db,slog.New(slog.DiscardHandler));api,ok:=any(r).(interface{CloseReportWorker(context.Context)error});if !ok{t.Fatal("bounded lifecycle unavailable")}
 req:=httptest.NewRequest(http.MethodPost,"/report",strings.NewReader(`{"provider_id":1,"ok":false}`));req.Header.Set("X-Control-Token","isolated-token");w:=httptest.NewRecorder();NewHTTPHandler(r,"isolated-token",slog.New(slog.DiscardHandler)).ServeHTTP(w,req)
 if w.Code!=http.StatusOK{t.Fatalf("expected current write-error characterization 200, got %d",w.Code)}
 states:=r.States();if len(states)!=1||states[0].ConsecutiveFail!=1{t.Fatalf("in-memory fail missing: %+v",states)}
 ctx,cancel:=context.WithTimeout(context.Background(),5*time.Second);defer cancel();if e:=api.CloseReportWorker(ctx);e!=nil{t.Fatal(e)}
}
