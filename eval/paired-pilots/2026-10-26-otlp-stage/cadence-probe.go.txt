type event struct { Name string `json:"name"`; MS float64 `json:"ms"`; OK bool `json:"ok"` }
type capture struct { mu sync.Mutex; start time.Time; events []event; endpoint string; base http.RoundTripper; requests int; Proxy bool `json:"proxy_selected"`; Status int `json:"http_status,omitempty"` }
func (c *capture) add(name string, ok bool) { c.mu.Lock(); defer c.mu.Unlock(); c.events=append(c.events,event{name,float64(time.Since(c.start))/float64(time.Millisecond),ok}) }
func(c *capture) RoundTrip(r *http.Request)(*http.Response,error) {
 c.mu.Lock(); c.requests++; n:=c.requests; c.mu.Unlock()
 if n>5 || r.Method!="POST" || r.URL.String()!=c.endpoint { c.add("unexpected_request_refused",false); return nil,errors.New("unexpected request refused") }
 proxy,e:=http.ProxyFromEnvironment(r); c.Proxy=proxy!=nil; c.add("proxy_lookup",e==nil)
 if e!=nil{return nil,errors.New("proxy lookup failed")}
 tr:=&httptrace.ClientTrace{GetConn:func(string){c.add("get_conn",true)},ConnectStart:func(string,string){c.add("connect_start",true)},ConnectDone:func(_, _ string,e error){c.add("connect_done",e==nil)},GotConn:func(httptrace.GotConnInfo){c.add("got_conn",true)},WroteRequest:func(x httptrace.WroteRequestInfo){c.add("wrote_request",x.Err==nil)},GotFirstResponseByte:func(){c.add("first_response_byte",true)}}
 out,e:=c.base.RoundTrip(r.Clone(httptrace.WithClientTrace(r.Context(),tr)));c.add("round_trip_return",e==nil)
 if out!=nil {c.Status=out.StatusCode};return out,e
}
