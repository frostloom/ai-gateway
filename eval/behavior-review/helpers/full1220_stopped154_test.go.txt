// Pure replay of pinned DATA originals. No recursive production constructor,
// keys, clients, settings, provider, DB, or Langfuse operation is invoked.
func stoppedV11OriginalBundle(t *testing.T) (windowActualManifest, map[string][]byte) {
	t.Helper()
	raw, e := os.ReadFile(windowStoppedV11InventoryPath)
	if e != nil {
		t.Fatal(e)
	}
	if conversationHash(raw) != windowStoppedV11InventorySHA {
		t.Fatal("descriptor changed")
	}
	var inv windowStoppedV11Inventory
	if e = json.Unmarshal(raw, &inv); e != nil {
		t.Fatal(e)
	}
	files := map[string][]byte{}
	for _, f := range inv.Files {
		if strings.HasPrefix(f.Path, "campaign/") || strings.HasPrefix(f.Path, "records/") || f.Path == "actual-manifest.json" || strings.HasPrefix(f.Path, "actual-owner-") {
			raw, e = os.ReadFile(filepath.Join(inv.Root, f.Path))
			if e != nil || conversationHash(raw) != f.SHA256 || int64(len(raw)) != f.Bytes {
				t.Fatalf("original %s: %v", f.Path, e)
			}
			files[f.Path] = raw
		} else {
			files[f.Path] = nil
		}
	}
	var m windowActualManifest
	if e = decodeWindowActualJSON(files["actual-manifest.json"], &m); e != nil {
		t.Fatal(e)
	}
	return m, files
}

// Snapshot the existing admitted pass metadata without changing admission,
// poisoning, fresh reads or End. This helper is outside production builds.
func stopped154DiagnosticSnapshot(p *historyReadPass) map[string]any {
	p.mu.Lock()
	defer p.mu.Unlock()
	classes := map[string]map[string]uint64{}
	class := func(path string) string {
		if historyInside(path, windowStoppedV11Root) {
			return "r3_original"
		}
		if historyInside(path, windowStoppedV11SourceRoot) {
			return "r2_source"
		}
		if path == windowStoppedV11InventoryPath {
			return "descriptor154"
		}
		return "predecessor_or_ancestor"
	}
	add := func(path, kind string, charge uint64) {
		name := class(path)
		if classes[name] == nil {
			classes[name] = map[string]uint64{}
		}
		classes[name][kind+"_records"]++
		classes[name][kind+"_charge"] += charge
	}
	charge := func(key, path, sha string, extra uint64) uint64 {
		return uint64(512+len(key)+len(path)+len(sha)) + extra
	}
	specsPerPath := map[string]int{}
	perPathCharge := map[string]uint64{}
	kinds := map[historyReadKind]uint64{}
	total := uint64(0)
	for key, e := range p.entries {
		c := charge(key, e.spec.path, e.spec.sha, 0)
		total += c
		add(e.spec.path, "entry", c)
		specsPerPath[e.spec.path]++
		perPathCharge[e.spec.path] += c
		kinds[e.spec.kind]++
	}
	for path := range p.dirs {
		c := charge("dir\x00"+path, path, "", 0)
		total += c
		add(path, "directory", c)
	}
	for path, ns := range p.namespaces {
		extra := uint64(0)
		for name := range ns.names {
			extra += uint64(24 + len(name))
		}
		c := charge("namespace\x00"+path, path, "", extra)
		total += c
		add(path, "namespace", c)
	}
	for path := range p.locks {
		c := charge("lock\x00"+path, path, "", 0)
		total += c
		add(path, "lock", c)
	}
	duplicates := []map[string]any{}
	for path, n := range specsPerPath {
		if n > 1 {
			duplicates = append(duplicates, map[string]any{"path_sha256": conversationHash([]byte(path)), "class": class(path), "specs": n, "charge": perPathCharge[path]})
		}
	}
	sort.Slice(duplicates, func(i, j int) bool {
		return duplicates[i]["path_sha256"].(string) < duplicates[j]["path_sha256"].(string)
	})
	return map[string]any{"schema": "stopped154-pass-diagnostic.v1", "stats": p.stats, "state": p.state, "active": p.active, "new154_metadata_active": p.stoppedV11Metadata, "legacy_transcript_bytes": p.stoppedV11LegacyTranscript, "limits": map[string]uint64{"transcript_total": map[bool]uint64{false: historyPassTranscriptLimit, true: 64 << 20}[p.stoppedV11Metadata], "transcript": historyPassTranscriptLimit, "records": historyPassRecordsLimit, "logical": historyPassLogicalLimit, "cache": historyPassCacheLimit}, "entries": len(p.entries), "unique_paths": len(p.paths), "directories": len(p.dirs), "namespaces": len(p.namespaces), "locks": len(p.locks), "entry_kind_counts": kinds, "charge_reconstructed": total, "charge_matches_stats": total == p.stats.TranscriptBytes, "classes": classes, "duplicate_spec_paths": duplicates}
}

func TestRootStopped154DiagnosticSnapshotPortable(t *testing.T) {
	root := stoppedEvidenceRoot(t)
	path := filepath.Join(root, "data")
	raw := []byte("data")
	if e := os.WriteFile(path, raw, 0600); e != nil {
		t.Fatal(e)
	}
	p, e := newHistoryReadPass(context.Background())
	if e != nil {
		t.Fatal(e)
	}
	defer p.Abort()
	for _, limit := range []int64{4, 8} {
		if _, e = historyStoppedV7File(p, path, conversationHash(raw), limit); e != nil {
			t.Fatal(e)
		}
	}
	s := stopped154DiagnosticSnapshot(p)
	if s["charge_matches_stats"] != true || s["entries"] != 2 || s["unique_paths"] != 1 || len(s["duplicate_spec_paths"].([]map[string]any)) != 1 {
		t.Fatalf("diagnostic metadata mismatch: %v", s)
	}
	if e = p.Finish(); e != nil {
		t.Fatal(e)
	}
}
