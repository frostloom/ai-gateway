type full1220SourceV12Job struct {
	Schema   string                 `json:"schema"`
	Boundary string                 `json:"observation_receipt_boundary"`
	Scope    fullWindowScope        `json:"scope"`
	Rows     []windowAcquisitionRow `json:"rows"`
	Binding  json.RawMessage        `json:"binding"`
}

func full1220ReplaySourceV12(job full1220SourceV12Job, dir string) error {
	source, err := LoadConversations("../../eval/task-definitions/2026-10-12-full-dev-v1")
	if err != nil {
		return err
	}
	profiles, err := LoadPairedProfiles("../../eval/runtime-profiles/2026-10-13-six-profiles-v1")
	if err != nil {
		return err
	}
	if job.Schema != "full1220-source-v12-job.v1" || (job.Boundary != "durable_local" && job.Boundary != "original_delivered_prefix") {
		return fmt.Errorf("explicit durable-local source boundary required")
	}
	if job.Scope.Bindings.PolicySHA256 != windowOwnedQualityPolicySHA() {
		return fmt.Errorf("source scope epoch/policy")
	}
	if job.Boundary == "original_delivered_prefix" {
		if job.Scope.Schema != "full-window-scope.v9" || windowV8DeclarationSHA(job.Scope) != windowV10OriginalScopeSHA {
			return fmt.Errorf("fixed original delivered-prefix source scope")
		}
		for _, row := range job.Rows {
			if row.Ordinal < 1 || row.Ordinal > 385 {
				return fmt.Errorf("prefix-only source row")
			}
		}
	} else if job.Scope.Schema != "full-window-scope.v12" {
		return fmt.Errorf("V12 suffix source scope")
	}
	if _, _, err = validateFullWindowScope(job.Scope, source, profiles, job.Scope.Bindings); err != nil {
		return err
	}
	primary, _, err := fullCustomerSelection(source)
	if err != nil {
		return err
	}
	cases := map[string]ConversationCase{}
	for _, c := range primary.Cases {
		cases[c.Key()] = c
	}
	seen := map[string]bool{}
	if len(job.Rows) == 0 || len(job.Rows) > 24 {
		return fmt.Errorf("bounded quality rows required")
	}
	first := job.Rows[0].Ordinal
	if first < 1 || first > len(job.Scope.Slots) {
		return fmt.Errorf("source first ordinal")
	}
	var window *fullWindowDeclaration
	for i := range job.Scope.Windows {
		if job.Scope.Windows[i].ID == job.Scope.Slots[first-1].WindowID {
			window = &job.Scope.Windows[i]
			break
		}
	}
	if window == nil || first != window.FirstOrdinal || len(job.Rows) != 2*(window.LastOrdinal-window.FirstOrdinal+1) {
		return fmt.Errorf("full current window rows required")
	}
	for position, row := range job.Rows {
		if row.Ordinal < 1 || row.Ordinal > len(job.Scope.Slots) {
			return fmt.Errorf("quality ordinal")
		}
		slot := job.Scope.Slots[row.Ordinal-1]
		if row.Ordinal != window.FirstOrdinal+position/2 || row.Arm != slot.ArmOrder[position%2] {
			return fmt.Errorf("source row schedule/order")
		}
		name := fmt.Sprintf("arm-%06d-%s.json", row.Ordinal, row.Arm)
		if seen[name] || row.RecordingFile != name || row.Pass != slot.Pass || row.CaseKey != slot.CaseKey || row.DefinedTurns != slot.UserTurns || (row.Arm != "model" && row.Arm != "jev_selective") || !row.Attempted || !row.Finished || !row.Cleanup || !row.Observer {
			return fmt.Errorf("quality slot/row mismatch")
		}
		seen[name] = true
		raw, err := readWindowActualFile(filepath.Join(dir, name), row.RecordingSHA256, 16<<20)
		if err != nil {
			return err
		}
		var record collectionRecording
		if err = decodeWindowStoppedV4QualityData(raw, &record); err != nil {
			return err
		}
		original, ok := cases[row.CaseKey]
		if !ok || !reflect.DeepEqual(record.Result.Case, original) || len(record.Result.Observations) != row.ObservedTurns || record.Result.Status != row.Status || record.Result.StructuralPass != row.StructuralPass || record.Result.Success != row.Success || record.Disposition != row.Disposition {
			return fmt.Errorf("quality frozen case/state mismatch")
		}
		for _, obs := range record.Result.Observations {
			tm := obs.Turn.Response.Telemetry
			if tm == nil || tm.Runtime == nil || tm.LangfuseTraceID != row.TraceID || tm.Runtime.RoutingMode != row.Arm {
				return fmt.Errorf("quality trace/arm mismatch")
			}
		}
		if row.QualityEnd == nil {
			if record.Disposition != "continue_recorded" || row.ObservedTurns != slot.UserTurns {
				return fmt.Errorf("ordinary completed row disposition/suffix")
			}
			if err = verifyCollectionRecording(record); err != nil {
				return err
			}
		} else {
			if !row.ExcludeCompleteTaskComparison {
				return fmt.Errorf("quality pair exclusion")
			}
			if _, err = readWindowStoppedV4Quality(dir, job.Scope.Bindings.PolicySHA256, slot, row, record); err != nil {
				return err
			}
		}
	}
	return nil
}

func TestFull1220SourceV12(t *testing.T) {
	input := os.Getenv("FULL1220_V12_SOURCE_INPUT")
	if input == "" {
		t.Skip("explicit offline quality replay only")
	}
	raw, err := readWindowActualFile(input, os.Getenv("FULL1220_V12_SOURCE_INPUT_SHA256"), 16<<20)
	if err != nil {
		t.Fatal(err)
	}
	var job full1220SourceV12Job
	if err = decodeWindowStoppedV4QualityData(raw, &job); err != nil {
		t.Fatal(err)
	}
	if err = full1220ReplaySourceV12(job, filepath.Join(filepath.Dir(input), "records")); err != nil {
		t.Fatal(err)
	}
	result, err := json.Marshal(struct {
		Schema      string          `json:"schema"`
		InputSHA256 string          `json:"input_sha256"`
		Binding     json.RawMessage `json:"binding"`
	}{"full1220-source-v12-replay.v1", conversationHash(raw), job.Binding})
	if err != nil {
		t.Fatal(err)
	}
	file, err := os.OpenFile(os.Getenv("FULL1220_V12_SOURCE_OUTPUT"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = file.Write(result); err != nil {
		file.Close()
		t.Fatal(err)
	}
	if err = file.Sync(); err != nil {
		file.Close()
		t.Fatal(err)
	}
	if err = file.Close(); err != nil {
		t.Fatal(err)
	}
	t.Logf("FULL1220_V12_SOURCE_OUTPUT_SHA256=%s", conversationHash(result))
}

// This command emits DATA only. The Python factory separately requires an
// authenticated normal owner exit and the complete511 source/native chain.
func TestFull1220SpoolV12(t *testing.T) {
	input := os.Getenv("FULL1220_V12_SPOOL_INPUT")
	if input == "" {
		t.Skip("explicit closed spool DATA replay only")
	}
	raw, err := readWindowActualFile(input, os.Getenv("FULL1220_V12_SPOOL_INPUT_SHA256"), 1<<20)
	if err != nil {
		t.Fatal(err)
	}
	var job struct {
		Schema   string `json:"schema"`
		Root     string `json:"root"`
		RunID    string `json:"run_id"`
		MaxBytes int64  `json:"max_bytes"`
	}
	if err = decodeWindowStoppedV4QualityData(raw, &job); err != nil || job.Schema != "full1220-v12-closed-spool-job.v1" {
		t.Fatal("spool job", err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 150*time.Second)
	defer cancel()
	proof, err := observability.VerifyClosedDurableSpool(ctx, observability.DurableSpoolConfig{Root: job.Root, RunID: job.RunID, MaxBytes: job.MaxBytes})
	if err != nil {
		t.Fatal(err)
	}
	// Read every batch through the opaque proof again before returning DATA.
	for i := uint64(1); i <= proof.Summary().Batches; i++ {
		if _, err = proof.ReadBatch(ctx, i); err != nil {
			t.Fatal(err)
		}
	}
	result, err := json.Marshal(struct {
		Schema      string                              `json:"schema"`
		InputSHA256 string                              `json:"input_sha256"`
		Summary     observability.DurableSpoolStatus    `json:"summary"`
		Spans       []observability.DurableSpanIdentity `json:"spans"`
	}{"full1220-v12-closed-spool-data.v1", conversationHash(raw), proof.Summary(), proof.SpanIdentities()})
	if err != nil {
		t.Fatal(err)
	}
	f, err := os.OpenFile(os.Getenv("FULL1220_V12_SPOOL_OUTPUT"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		t.Fatal(err)
	}
	if n, e := f.Write(result); e != nil || n != len(result) {
		f.Close()
		t.Fatal("spool output write", e)
	}
	if err = f.Sync(); err != nil {
		f.Close()
		t.Fatal(err)
	}
	if err = f.Close(); err != nil {
		t.Fatal(err)
	}
	t.Logf("FULL1220_V12_SPOOL_OUTPUT_SHA256=%s", conversationHash(result))
}

func TestFull1220V12CodecVector(t *testing.T) {
	// Read only synthetic DATA: no private key, signing or actual constructor.
	raw, err := os.ReadFile("../../eval/behavior-review/fixtures/v12-contracts/go-authority-data.json")
	if err != nil {
		t.Fatal(err)
	}
	var vector struct {
		DataOnly      bool                            `json:"data_only"`
		Authorization windowV12ExecutionAuthorization `json:"authorization"`
		SigningBytes  string                          `json:"signing_bytes"`
	}
	if err = json.Unmarshal(raw, &vector); err != nil {
		t.Fatal(err)
	}
	if !vector.DataOnly || vector.Authorization.Signature != strings.Repeat("0", 128) {
		t.Fatal("synthetic DATA required")
	}
	canonical, err := json.Marshal(vector.Authorization.windowV12ExecutionBinding)
	if err != nil {
		t.Fatal(err)
	}
	if string(canonical) != vector.SigningBytes {
		t.Fatal("canonical DATA vector mismatch")
	}
	scope := full1220V12HelperTestScope(t)
	authority, err := json.Marshal(vector.Authorization)
	if err != nil {
		t.Fatal(err)
	}
	if os.Getenv("FULL1220_V12_VECTOR_OUTPUT") != "" {
		raw, err := json.Marshal(map[string]any{"evidence_kind": "synthetic_go_codec_data_not_authority", "authorization": json.RawMessage(authority), "canonical_utf8": string(canonical), "budget": expectedWindowV12BudgetDeclaration(), "budget_sha256": windowV8DeclarationSHA(expectedWindowV12BudgetDeclaration()), "scope": scope})
		if err != nil {
			t.Fatal(err)
		}
		f, err := os.OpenFile(os.Getenv("FULL1220_V12_VECTOR_OUTPUT"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
		if err != nil {
			t.Fatal(err)
		}
		if _, err = f.Write(raw); err != nil {
			f.Close()
			t.Fatal(err)
		}
		if err = f.Close(); err != nil {
			t.Fatal(err)
		}
	}
}

func TestFull1220V12SourceRejectsIncompleteAndOldBoundary(t *testing.T) {
	scope := full1220V12HelperTestScope(t)
	scope.Bindings.PolicySHA256 = windowOwnedQualityPolicySHA()
	for _, job := range []full1220SourceV12Job{
		{Schema: "full1220-source-v12-job.v1", Boundary: "durable_local", Scope: scope},
		{Schema: "full1220-source-v12-job.v1", Boundary: "remote", Scope: scope},
		{Schema: "full1220-source-v9-job.v1", Boundary: "durable_local", Scope: scope},
	} {
		if full1220ReplaySourceV12(job, t.TempDir()) == nil {
			t.Fatal("incomplete/old source job accepted")
		}
	}
}

func TestFull1220V12ClosedSpoolHelperRoundTrip(t *testing.T) {
	o, dc := full1220V12HelperTestObserver(t)
	ctx := context.Background()
	_, turn := o.StartTurn(ctx, "model")
	turn.End(200, observability.Summary{})
	if e := o.Shutdown(ctx); e != nil {
		t.Fatal(e)
	}
	dir := stoppedEvidenceRoot(t)
	input := filepath.Join(dir, "input.json")
	output := filepath.Join(dir, "output.json")
	raw, _ := json.Marshal(map[string]any{"schema": "full1220-v12-closed-spool-job.v1", "root": dc.Root, "run_id": dc.RunID, "max_bytes": dc.MaxBytes})
	if e := os.WriteFile(input, raw, 0600); e != nil {
		t.Fatal(e)
	}
	t.Setenv("FULL1220_V12_SPOOL_INPUT", input)
	t.Setenv("FULL1220_V12_SPOOL_INPUT_SHA256", conversationHash(raw))
	t.Setenv("FULL1220_V12_SPOOL_OUTPUT", output)
	TestFull1220SpoolV12(t)
	result, e := os.ReadFile(output)
	if e != nil {
		t.Fatal(e)
	}
	var data struct {
		Schema      string                              `json:"schema"`
		InputSHA256 string                              `json:"input_sha256"`
		Summary     observability.DurableSpoolStatus    `json:"summary"`
		Spans       []observability.DurableSpanIdentity `json:"spans"`
	}
	if e = json.Unmarshal(result, &data); e != nil {
		t.Fatal(e)
	}
	if !data.Summary.Closed || !data.Summary.DataOnly || !data.Summary.DurabilityReturnUnproven || !data.Summary.OwnerCompletionUnproven || data.Summary.Spans != 1 || len(data.Spans) != 1 || data.InputSHA256 != conversationHash(raw) {
		t.Fatal("DATA boundary changed")
	}
}
