func full1220ReplaySourceV15(job full1220SourceV15Job, dir string) error {
	source, err := LoadConversations("../../eval/task-definitions/2026-10-12-full-dev-v1")
	if err != nil {
		return err
	}
	profiles, err := LoadPairedProfiles("../../eval/runtime-profiles/2026-10-13-six-profiles-v1")
	if err != nil {
		return err
	}
	if job.Schema != "full1220-source-v15-job.v1" || (job.Boundary != "durable_local" && job.Boundary != "original_delivered_prefix") {
		return fmt.Errorf("explicit durable-local source boundary required")
	}
	if job.Boundary == "original_delivered_prefix" {
		if job.Scope.Bindings.PolicySHA256 != windowOwnedQualityPolicySHA() {
			return fmt.Errorf("original source policy")
		}
		if job.Scope.Schema != "full-window-scope.v9" || windowV8DeclarationSHA(job.Scope) != windowV10OriginalScopeSHA {
			return fmt.Errorf("fixed original delivered-prefix source scope")
		}
		for _, row := range job.Rows {
			if row.Ordinal < 1 || row.Ordinal > 385 {
				return fmt.Errorf("prefix-only source row")
			}
		}
	} else if job.Scope.Schema != "full-window-scope.v15" || job.Scope.CollectionPolicySHA256 != collectionPreviewFailurePolicySHA() || job.Scope.Bindings.PolicySHA256 != windowOwnedPreviewFailurePolicySHA() {
		return fmt.Errorf("V15 suffix source scope")
	}
	if _, _, err = validateFullWindowScope(job.Scope, source, profiles, job.Scope.Bindings); err != nil {
		return err
	}
	primary, _, err := fullCustomerSelection(source)
	if err != nil {
		return err
	}
	cases := map[string]ConversationCase{}
	for _, c := range primary.Cases {
		cases[c.Key()] = c
	}
	seen := map[string]bool{}
	if len(job.Rows) == 0 || len(job.Rows) > 24 {
		return fmt.Errorf("bounded quality rows required")
	}
	first := job.Rows[0].Ordinal
	if first < 1 || first > len(job.Scope.Slots) {
		return fmt.Errorf("source first ordinal")
	}
	var window *fullWindowDeclaration
	for i := range job.Scope.Windows {
		if job.Scope.Windows[i].ID == job.Scope.Slots[first-1].WindowID {
			window = &job.Scope.Windows[i]
			break
		}
	}
	if window == nil || first != window.FirstOrdinal || len(job.Rows) != 2*(window.LastOrdinal-window.FirstOrdinal+1) {
		return fmt.Errorf("full current window rows required")
	}
	for position, row := range job.Rows {
		if row.Ordinal < 1 || row.Ordinal > len(job.Scope.Slots) {
			return fmt.Errorf("quality ordinal")
		}
		slot := job.Scope.Slots[row.Ordinal-1]
		if row.Ordinal != window.FirstOrdinal+position/2 || row.Arm != slot.ArmOrder[position%2] {
			return fmt.Errorf("source row schedule/order")
		}
		name := fmt.Sprintf("arm-%06d-%s.json", row.Ordinal, row.Arm)
		if seen[name] || row.RecordingFile != name || row.Pass != slot.Pass || row.CaseKey != slot.CaseKey || row.DefinedTurns != slot.UserTurns || (row.Arm != "model" && row.Arm != "jev_selective") || !row.Attempted || !row.Finished || !row.Cleanup || !row.Observer {
			return fmt.Errorf("quality slot/row mismatch")
		}
		seen[name] = true
		raw, err := readWindowActualFile(filepath.Join(dir, name), row.RecordingSHA256, 16<<20)
		if err != nil {
			return err
		}
		var record collectionRecording
		if err = decodeWindowStoppedV4QualityData(raw, &record); err != nil {
			return err
		}
		original, ok := cases[row.CaseKey]
		if !ok || !reflect.DeepEqual(record.Result.Case, original) || len(record.Result.Observations) != row.ObservedTurns || record.Result.Status != row.Status || record.Result.StructuralPass != row.StructuralPass || record.Result.Success != row.Success || record.Disposition != row.Disposition {
			return fmt.Errorf("quality frozen case/state mismatch")
		}
		for _, obs := range record.Result.Observations {
			tm := obs.Turn.Response.Telemetry
			if tm == nil || tm.Runtime == nil || tm.LangfuseTraceID != row.TraceID || tm.Runtime.RoutingMode != row.Arm {
				return fmt.Errorf("quality trace/arm mismatch")
			}
		}
		if row.QualityEnd == nil {
			if record.Disposition != "continue_recorded" || row.ObservedTurns != slot.UserTurns {
				return fmt.Errorf("ordinary completed row disposition/suffix")
			}
			if job.Boundary == "original_delivered_prefix" {
				if row.PreviewFailure != nil || verifyCollectionRecording(record) != nil {
					return fmt.Errorf("original recording/sidecar policy")
				}
			} else {
				if row.PreviewFailure == nil || row.PreviewFailure.SidecarFile != fmt.Sprintf("preview-failure-%06d-%s.json", row.Ordinal, row.Arm) {
					return fmt.Errorf("V15 normal row sidecar required")
				}
				sidecarRaw, err := readWindowActualFile(filepath.Join(dir, row.PreviewFailure.SidecarFile), row.PreviewFailure.SidecarSHA256, 32<<20)
				if err != nil || verifyWindowPreviewFailureSidecar(sidecarRaw, record, row.PreviewFailure) != nil {
					return fmt.Errorf("V15 failed-preview source replay")
				}
			}
		} else {
			if !row.ExcludeCompleteTaskComparison {
				return fmt.Errorf("quality pair exclusion")
			}
			if row.PreviewFailure != nil {
				return fmt.Errorf("quality row cannot carry normal sidecar")
			}
			if job.Boundary == "original_delivered_prefix" {
				_, err = readWindowStoppedV4Quality(dir, job.Scope.Bindings.PolicySHA256, slot, row, record)
			} else {
				_, err = full1220ReadV15Quality(dir, job.Scope.Bindings.PolicySHA256, slot, row, record)
			}
			if err != nil {
				return err
			}
		}
	}
	return nil
}

func full1220ReadV15Quality(dir, policy string, slot fullWindowSlot, row windowAcquisitionRow, record collectionRecording) (*windowQualityOwnedEnd, error) {
	reject := func() (*windowQualityOwnedEnd, error) {
		return nil, fmt.Errorf("stopped v4 quality sidecar linkage rejected")
	}
	q := row.QualityEnd
	if validateWindowQualityEndEvidence(policy, slot, row.Arm, row.ObservedTurns, q) != nil || row.Status != "failed" || row.StructuralPass || row.Success || record.Disposition != "halt" || q.CandidateFile != fmt.Sprintf("quality-%06d-%s-candidate.json", row.Ordinal, row.Arm) || q.CleanupFile != fmt.Sprintf("quality-%06d-%s-cleanup.json", row.Ordinal, row.Arm) {
		return reject()
	}
	candidateRaw, err := readWindowActualFile(filepath.Join(dir, q.CandidateFile), q.CandidateSHA256, 16<<20)
	if err != nil {
		return reject()
	}
	var candidate qualityPreviewSidecar
	if decodeWindowStoppedV4QualityData(candidateRaw, &candidate) != nil || candidate.PolicyID != "full-customer-owned-preview-candidate.v13" || verifyWindowQualityCandidate(candidate) != nil || !reflect.DeepEqual(candidate.Evidence.Record, record) || candidate.SourceSHA256 != q.SourceSHA256 || candidate.RuleSHA256 != q.RuleSHA256 || len(candidate.Unrun) != q.UnrunTurns {
		return reject()
	}
	cleanupRaw, err := readWindowActualFile(filepath.Join(dir, q.CleanupFile), q.CleanupSHA256, 16<<20)
	if err != nil {
		return reject()
	}
	var cleanup struct {
		Post    qualityEffectCapture `json:"post_cleanup"`
		Deleted int64                `json:"deleted"`
		Absent  bool                 `json:"session_absent"`
	}
	if decodeWindowStoppedV4QualityData(cleanupRaw, &cleanup) != nil || cleanup.Deleted != 1 || !cleanup.Absent || !reflect.DeepEqual(cleanup.Post, candidate.Evidence.After) {
		return reject()
	}
	owned := &windowQualityOwnedEnd{Evidence: *q, Candidate: candidate, PostCleanup: cleanup.Post, Deleted: cleanup.Deleted, SessionAbsent: cleanup.Absent}
	if windowQualityEnvelopeSHA(owned) != q.EnvelopeSHA256 {
		return reject()
	}
	return owned, nil
}

func TestFull1220SourceV15(t *testing.T) {
	input := os.Getenv("FULL1220_V15_SOURCE_INPUT")
	if input == "" {
		t.Skip("explicit offline quality replay only")
	}
	raw, err := readWindowActualFile(input, os.Getenv("FULL1220_V15_SOURCE_INPUT_SHA256"), 16<<20)
	if err != nil {
		t.Fatal(err)
	}
	var job full1220SourceV15Job
	if err = decodeWindowStoppedV4QualityData(raw, &job); err != nil {
		t.Fatal(err)
	}
	if err = full1220ReplaySourceV15(job, filepath.Join(filepath.Dir(input), "records")); err != nil {
		t.Fatal(err)
	}
	result, err := json.Marshal(struct {
		Schema      string          `json:"schema"`
		InputSHA256 string          `json:"input_sha256"`
		Binding     json.RawMessage `json:"binding"`
	}{"full1220-source-v15-replay.v1", conversationHash(raw), job.Binding})
	if err != nil {
		t.Fatal(err)
	}
	file, err := os.OpenFile(os.Getenv("FULL1220_V15_SOURCE_OUTPUT"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = file.Write(result); err != nil {
		file.Close()
		t.Fatal(err)
	}
	if err = file.Sync(); err != nil {
		file.Close()
		t.Fatal(err)
	}
	if err = file.Close(); err != nil {
		t.Fatal(err)
	}
	t.Logf("FULL1220_V15_SOURCE_OUTPUT_SHA256=%s", conversationHash(result))
}

// This command emits DATA only. The Python factory separately requires an
// authenticated normal owner exit and the complete456 suffix and original55 source/native chain.
func TestFull1220SpoolV15(t *testing.T) {
	input := os.Getenv("FULL1220_V15_SPOOL_INPUT")
	if input == "" {
		t.Skip("explicit closed spool DATA replay only")
	}
	raw, err := readWindowActualFile(input, os.Getenv("FULL1220_V15_SPOOL_INPUT_SHA256"), 1<<20)
	if err != nil {
		t.Fatal(err)
	}
	var job struct {
		Schema   string `json:"schema"`
		Root     string `json:"root"`
		RunID    string `json:"run_id"`
		MaxBytes int64  `json:"max_bytes"`
	}
	if err = decodeWindowStoppedV4QualityData(raw, &job); err != nil || job.Schema != "full1220-v15-closed-spool-job.v1" {
		t.Fatal("spool job", err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 210*time.Second)
	defer cancel()
	proof, err := observability.VerifyClosedDurableSpool(ctx, observability.DurableSpoolConfig{Root: job.Root, RunID: job.RunID, MaxBytes: job.MaxBytes})
	if err != nil {
		t.Fatal(err)
	}
	// Read every batch through the opaque proof again before returning DATA.
	for i := uint64(1); i <= proof.Summary().Batches; i++ {
		if _, err = proof.ReadBatch(ctx, i); err != nil {
			t.Fatal(err)
		}
	}
	result, err := json.Marshal(struct {
		Schema      string                              `json:"schema"`
		InputSHA256 string                              `json:"input_sha256"`
		Summary     observability.DurableSpoolStatus    `json:"summary"`
		Spans       []observability.DurableSpanIdentity `json:"spans"`
	}{"full1220-v15-closed-spool-data.v1", conversationHash(raw), proof.Summary(), proof.SpanIdentities()})
	if err != nil {
		t.Fatal(err)
	}
	f, err := os.OpenFile(os.Getenv("FULL1220_V15_SPOOL_OUTPUT"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		t.Fatal(err)
	}
	if n, e := f.Write(result); e != nil || n != len(result) {
		f.Close()
		t.Fatal("spool output write", e)
	}
	if err = f.Sync(); err != nil {
		f.Close()
		t.Fatal(err)
	}
	if err = f.Close(); err != nil {
		t.Fatal(err)
	}
	t.Logf("FULL1220_V15_SPOOL_OUTPUT_SHA256=%s", conversationHash(result))
}

// Canonical V15 execution/policy-approval wire DATA is covered by the original
// Go-produced fixtures/v15-contracts/go-operation-wire-data.json and external
// codec tests. This helper does not duplicate the obsolete V13 vector
// shape or produce any signed authorization/permission artifact.

func TestFull1220V15SourceRejectsIncompleteAndOldBoundary(t *testing.T) {
	scope := full1220V15HelperTestScope(t)
	scope.Bindings.PolicySHA256 = windowOwnedPreviewFailurePolicySHA()
	old := scope
	old.Schema = "full-window-scope.v14"
	for _, job := range []full1220SourceV15Job{
		{Schema: "full1220-source-v15-job.v1", Boundary: "durable_local", Scope: scope},
		{Schema: "full1220-source-v15-job.v1", Boundary: "remote", Scope: scope},
		{Schema: "full1220-source-v9-job.v1", Boundary: "durable_local", Scope: scope},
		{Schema: "full1220-source-v14-job.v1", Boundary: "durable_local", Scope: scope},
		{Schema: "full1220-source-v15-job.v1", Boundary: "durable_local", Scope: old},
	} {
		if full1220ReplaySourceV15(job, t.TempDir()) == nil {
			t.Fatal("incomplete/old source job accepted")
		}
	}
}

func TestFull1220V15ClosedSpoolHelperRoundTrip(t *testing.T) {
	o, dc := full1220V15HelperTestObserver(t)
	ctx := context.Background()
	_, turn := o.StartTurn(ctx, "model")
	turn.End(200, observability.Summary{})
	if e := o.Shutdown(ctx); e != nil {
		t.Fatal(e)
	}
	dir := stoppedEvidenceRoot(t)
	input := filepath.Join(dir, "input.json")
	output := filepath.Join(dir, "output.json")
	raw, _ := json.Marshal(map[string]any{"schema": "full1220-v15-closed-spool-job.v1", "root": dc.Root, "run_id": dc.RunID, "max_bytes": dc.MaxBytes})
	if e := os.WriteFile(input, raw, 0600); e != nil {
		t.Fatal(e)
	}
	t.Setenv("FULL1220_V15_SPOOL_INPUT", input)
	t.Setenv("FULL1220_V15_SPOOL_INPUT_SHA256", conversationHash(raw))
	t.Setenv("FULL1220_V15_SPOOL_OUTPUT", output)
	TestFull1220SpoolV15(t)
	result, e := os.ReadFile(output)
	if e != nil {
		t.Fatal(e)
	}
	var data struct {
		Schema      string                              `json:"schema"`
		InputSHA256 string                              `json:"input_sha256"`
		Summary     observability.DurableSpoolStatus    `json:"summary"`
		Spans       []observability.DurableSpanIdentity `json:"spans"`
	}
	if e = json.Unmarshal(result, &data); e != nil {
		t.Fatal(e)
	}
	if !data.Summary.Closed || !data.Summary.DataOnly || !data.Summary.DurabilityReturnUnproven || !data.Summary.OwnerCompletionUnproven || data.Summary.Spans != 1 || len(data.Spans) != 1 || data.InputSHA256 != conversationHash(raw) {
		t.Fatal("DATA boundary changed")
	}
}
